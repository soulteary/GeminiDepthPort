cmake_minimum_required(VERSION 3.16)
project(gdp_sdk_diagnostic_experiment LANGUAGES C CXX)
find_package(Threads REQUIRED)
if(NOT IS_DIRECTORY "${SDK_SOURCE}/src")
  message(FATAL_ERROR "SDK_SOURCE must point to SDK 2.10.6 source")
endif()
set(OB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OB_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(OB_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(OB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OB_INSTALL_EXAMPLES_SOURCE OFF CACHE BOOL "" FORCE)
set(OB_BUILD_WITH_EXTENSIONS_COMMIT_HASH OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDK_SOURCE}" sdk)
# Link the same implementation/archives as the actual SDK, allowing an in-memory
# IDevice fixture. -fno-access-control is test-driver-only, never an SDK flag.
get_target_property(SDK_IMPL OrbbecSDK SOURCES)
get_target_property(SDK_LINKS OrbbecSDK LINK_LIBRARIES)
add_executable(gdp_sdk_probe probe.cpp ${SDK_IMPL})
target_include_directories(gdp_sdk_probe PRIVATE "${SDK_SOURCE}/src" "${CMAKE_BINARY_DIR}/sdk/src/generated")
target_link_libraries(gdp_sdk_probe PRIVATE ${SDK_LINKS})
target_compile_options(gdp_sdk_probe PRIVATE -fno-access-control)
target_compile_definitions(gdp_sdk_probe PRIVATE OB_SDK_LIBRARY_NAME="OrbbecSDK")
set_target_properties(gdp_sdk_probe PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(EXISTS "${SDK_SOURCE}/src/shared/utils/CaptureDiagnostics.hpp")
  target_compile_definitions(gdp_sdk_probe PRIVATE GDP_PATCHED=1)
  add_executable(gdp_diagnostic_unit diagnostic_unit.cpp)
  target_include_directories(gdp_diagnostic_unit PRIVATE "${SDK_SOURCE}/src/shared")
  target_link_libraries(gdp_diagnostic_unit PRIVATE pthread)
  target_compile_options(gdp_diagnostic_unit PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  set_target_properties(gdp_diagnostic_unit PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
endif()
