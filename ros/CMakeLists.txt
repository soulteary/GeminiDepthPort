cmake_minimum_required(VERSION 3.16)
project(geminidepthport_ros_bridge LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
add_library(geminidepthport_ros_bridge SHARED src/bridge.cpp src/auxiliary.cpp src/imu.cpp src/context.cpp)
target_compile_features(geminidepthport_ros_bridge PRIVATE cxx_std_17)
target_compile_options(geminidepthport_ros_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(geminidepthport_ros_bridge PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(geminidepthport_ros_bridge rclcpp sensor_msgs std_msgs geometry_msgs tf2_msgs)
install(TARGETS geminidepthport_ros_bridge LIBRARY DESTINATION lib)
install(FILES include/gdp_ros_bridge.h DESTINATION include)
install(FILES include/gdp_ros_imu.h include/gdp_ros_context.h DESTINATION include)
install(FILES include/gdp_ros_aux.h DESTINATION include)
install(PROGRAMS tests/smoke_ros.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS tests/smoke_context.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS tests/smoke_imu.py DESTINATION lib/${PROJECT_NAME})
install(FILES imu_recording.py DESTINATION lib)
install(PROGRAMS tests/smoke_aux.py DESTINATION lib/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_libraries(geminidepthport_ros_bridge)
ament_export_dependencies(rclcpp sensor_msgs std_msgs geometry_msgs tf2_msgs)
ament_package()
